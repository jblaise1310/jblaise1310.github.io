-- Create new_med_db table for medical data
CREATE TABLE new_med_db (
    customer_id VARCHAR(50),
    state VARCHAR(2),
    age INT,
    gender VARCHAR(10),
    marital VARCHAR(20),
    children INT,
    income FLOAT,
    allergic_rhinitis BOOLEAN,
    anxiety BOOLEAN,
    asthma BOOLEAN,
    arthritis BOOLEAN,
    backpain BOOLEAN,
    diabetes BOOLEAN,
    highblood BOOLEAN,
    hyperlipidemia BOOLEAN,
    overweight BOOLEAN,
    reflux_esophagitis BOOLEAN,
    stroke BOOLEAN,
    complication_risk VARCHAR(10),
    readmis BOOLEAN
);

-- Create new_sdoh_db table for SDoH data
CREATE TABLE new_sdoh_db (
    sdoh_id SERIAL PRIMARY KEY,
    state VARCHAR(2),
    total_population INT,
    measure VARCHAR(255),
    data_value FLOAT,
    sdoh_factors VARCHAR(255),
    sdoh_score FLOAT
);

-- Add score columns to new_med_db for medical risk scores
ALTER TABLE new_med_db
ADD COLUMN allergic_rhinitis_score FLOAT,
ADD COLUMN anxiety_score FLOAT,
ADD COLUMN asthma_score FLOAT,
ADD COLUMN arthritis_score FLOAT,
ADD COLUMN backpain_score FLOAT,
ADD COLUMN diabetes_score FLOAT,
ADD COLUMN highblood_score FLOAT,
ADD COLUMN hyperlipidemia_score FLOAT,
ADD COLUMN overweight_score FLOAT,
ADD COLUMN reflux_esophagitis_score FLOAT,
ADD COLUMN stroke_score FLOAT,
ADD COLUMN readmis_score FLOAT,
ADD COLUMN complication_risk_score FLOAT;

-- Update scores based on conditions
UPDATE new_med_db
SET allergic_rhinitis_score = CASE WHEN allergic_rhinitis = 'Yes' THEN 2 ELSE 0 END,
    anxiety_score = CASE WHEN anxiety = 'Yes' THEN 2 ELSE 0 END,
    asthma_score = CASE WHEN asthma = 'Yes' THEN 6 ELSE 0 END,
    arthritis_score = CASE WHEN arthritis = 'Yes' THEN 6 ELSE 0 END,
    backpain_score = CASE WHEN backpain = 'Yes' THEN 4 ELSE 0 END,
    diabetes_score = CASE WHEN diabetes = 'Yes' THEN 8 ELSE 0 END,
    highblood_score = CASE WHEN highblood = 'Yes' THEN 8 ELSE 0 END,
    hyperlipidemia_score = CASE WHEN hyperlipidemia = 'Yes' THEN 4 ELSE 0 END,
    overweight_score = CASE WHEN overweight = 'Yes' THEN 3 ELSE 0 END,
    reflux_esophagitis_score = CASE WHEN reflux_esophagitis = 'Yes' THEN 3 ELSE 0 END,
    stroke_score = CASE WHEN stroke = 'Yes' THEN 10 ELSE 0 END,
    readmis_score = CASE WHEN readmis = 'Yes' THEN 3 ELSE 0 END,
    complication_risk_score = CASE 
        WHEN complication_risk = 'Low' THEN 1 
        WHEN complication_risk = 'Medium' THEN 3 
        WHEN complication_risk = 'High' THEN 6 
        ELSE 0 END;

-- Add and calculate Medical_risk_score
ALTER TABLE new_med_db ADD COLUMN Medical_risk_score INT;

UPDATE new_med_db
SET Medical_risk_score = 
    COALESCE(allergic_rhinitis_score, 0) +
    COALESCE(anxiety_score, 0) +
    COALESCE(asthma_score, 0) +
    COALESCE(arthritis_score, 0) +
    COALESCE(backpain_score, 0) +
    COALESCE(diabetes_score, 0) +
    COALESCE(highblood_score, 0) +
    COALESCE(hyperlipidemia_score, 0) +
    COALESCE(overweight_score, 0) +
    COALESCE(reflux_esophagitis_score, 0) +
    COALESCE(stroke_score, 0) +
    COALESCE(complication_risk_score, 0);

-- Add coding for referential integrity
ALTER TABLE new_med_db
ADD COLUMN sdoh_id INT;
